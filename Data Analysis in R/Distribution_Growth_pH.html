sample_info_no_R <-read.table("Alaska_info_no_R.txt", row.names=1, header=T, check.names=F)

df_correlation <- sample_info_no_R[, c("BAI_5y","BAI_10y","BAI_15y","Age", "DBH", "Height", "pH")]
df_correlation <- na.omit(df_correlation)

# Compute correlations + p-values
cor_results <- rcorr(as.matrix(df_correlation))
cor_matrix <- cor_results$r
p_matrix <- cor_results$P

sig_matrix <- p_matrix < 0.05

lower_fun_sig <- function(data, mapping, ...) {
  var_x <- rlang::as_name(mapping$x)
  var_y <- rlang::as_name(mapping$y)
if (var_x %in% rownames(sig_matrix) && var_y %in% colnames(sig_matrix)) {
    sig <- sig_matrix[var_x, var_y]
} else if (var_y %in% rownames(sig_matrix) && var_x %in% colnames(sig_matrix)) {
    sig <- sig_matrix[var_y, var_x]
} else {sig <- NA
}
point_color <- ifelse(isTRUE(sig), "steelblue", "gray70")
n_complete <- sum(complete.cases(data[[var_x]], data[[var_y]]))
p <- ggplot(data = data, mapping = mapping) +
    geom_point(alpha = 0.7, color = point_color, size = 1.5)
if (n_complete >= 3) {
p <- p + geom_smooth(method = "lm", se = FALSE, color = "darkred", size = 0.7)}
return(p)}

ggpairs(df_correlation,
        lower = list(continuous = lower_fun_sig),
